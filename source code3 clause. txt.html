/* -----------------------------------------------
   STATE & NAVIGATION
----------------------------------------------- */
const TOTAL_Q = 5;
let cls = '9', chapter = '';
let qIndex = 0, score = 0, xp = 0, level = 1, streak = 0;
let answered = false;
let usedIndices = [];

function goToQuiz() {
  document.getElementById('frontPage').classList.remove('active');
  document.getElementById('quizPage').classList.add('active');
  loadChapters();
}

function goHome() {
  document.getElementById('quizPage').classList.remove('active');
  document.getElementById('frontPage').classList.add('active');
}

/* -----------------------------------------------
   CLASS & CHAPTER
----------------------------------------------- */
function setClass(c, btn) {
  cls = c;
  document.querySelectorAll('.cls-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  loadChapters();
}

function loadChapters() {
  const sel = document.getElementById('chapterSel');
  sel.innerHTML = '';
  Object.keys(QB[cls]).forEach(ch => {
    const o = document.createElement('option');
    o.value = ch; o.textContent = ch;
    sel.appendChild(o);
  });
  chapter = Object.keys(QB[cls])[0];
  restartQuiz();
}

function changeChapter() {
  chapter = document.getElementById('chapterSel').value;
  restartQuiz();
}

/* -----------------------------------------------
   QUIZ LOGIC
----------------------------------------------- */
function restartQuiz() {
  qIndex = 0; score = 0; streak = 0;
  usedIndices = [];
  updateStats();
  document.getElementById('resultEl').style.display = 'none';
  document.getElementById('quizContent').style.display = 'block';
  document.getElementById('feedback').textContent = '';
  showQuestion();
}

function getRandomQuestion() {
  const pool = QB[cls][chapter];
  const available = pool.map((_, i) => i).filter(i => !usedIndices.includes(i));
  if (available.length === 0) usedIndices = [];
  const pick = available.length > 0
    ? available[Math.floor(Math.random() * available.length)]
    : Math.floor(Math.random() * pool.length);
  usedIndices.push(pick);
  return pool[pick];
}

function showQuestion() {
  answered = false;
  document.getElementById('nextBtn').classList.remove('show');
  document.getElementById('feedback').textContent = '';

  const q = getRandomQuestion();
  const labels = ['A', 'B', 'C', 'D'];

  document.getElementById('qCount').textContent = `Question ${qIndex + 1} of ${TOTAL_Q}`;
  document.getElementById('chapterTag').textContent = chapter;
  document.getElementById('qText').textContent = q.q;

  const el = document.getElementById('optionsEl');
  el.innerHTML = '';
  q.options.forEach((opt, i) => {
    const btn = document.createElement('button');
    btn.className = 'option-btn';
    btn.innerHTML = `<span class="opt-letter">${labels[i]}</span><span>${opt}</span>`;
    btn.onclick = () => checkAnswer(i, q.answer);
    el.appendChild(btn);
  });
}

function checkAnswer(chosen, correct) {
  if (answered) return;
  answered = true;

  document.querySelectorAll('.option-btn').forEach((b, i) => {
    b.disabled = true;
    if (i === correct) b.classList.add('correct');
    if (i === chosen && chosen !== correct) b.classList.add('wrong');
  });

  const fb = document.getElementById('feedback');
  if (chosen === correct) {
    fb.textContent = '? Correct!';
    fb.className = 'feedback ok';
    score += 10;
    xp += 20;
    streak++;
    if (streak >= 3) xp += 5;
    if (xp >= level * 100) {
      level++;
      showToast('? Level Up! Now Level ' + level);
    }
    bumpStat('scoreVal');
    if (streak > 1) bumpStat('streakVal');
  } else {
    fb.textContent = '? Wrong � correct answer is highlighted.';
    fb.className = 'feedback err';
    streak = 0;
  }

  updateStats();
  document.getElementById('nextBtn').classList.add('show');
}

function nextQ() {
  qIndex++;
  if (qIndex >= TOTAL_Q) {
    showResult();
  } else {
    showQuestion();
  }
}

function showResult() {
  document.getElementById('quizContent').style.display = 'none';
  document.getElementById('resultEl').style.display = 'block';
  const pct = Math.round(score / (TOTAL_Q * 10) * 100);
  const icon = pct >= 80 ? '??' : pct >= 60 ? '??' : pct >= 40 ? '??' : '??';
  const title = pct >= 80 ? 'Excellent!' : pct >= 60 ? 'Good job!' : pct >= 40 ? 'Keep going!' : 'Keep practicing!';
  document.getElementById('resultIcon').textContent = icon;
  document.getElementById('resultTitle').textContent = title;
  document.getElementById('resultScore').textContent = pct + '%';
  document.getElementById('resultSub').textContent = score + ' / ' + (TOTAL_Q * 10) + ' points � ' + chapter;
}

function updateStats() {
  document.getElementById('scoreVal').textContent = score;
  document.getElementById('levelVal').textContent = level;

  document.getElementById('xpLevel').textContent = 'Level ' + level;

  const cur = xp % need;
  document.getElementById('xpNums').textContent = cur + ' / ' + need + ' XP';

}

function bumpStat(id) {
  const el = document.getElementById(id);
  el.classList.add('bump');
  setTimeout(() => el.classList.remove('bump'), 700);
}
